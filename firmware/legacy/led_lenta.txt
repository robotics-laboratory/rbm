#include <Arduino.h>
#include <FastLED.h>

#define WS_LED_PIN 16
#define NUM_LEDS 16
CRGB leds[NUM_LEDS];

void setup() { 
  FastLED.addLeds<WS2812, WS_LED_PIN, GRB>(leds, NUM_LEDS); 
  FastLED.setBrightness(50);
}

void loop() {
    fill_solid(leds, NUM_LEDS, CRGB(255, 255, 255)); FastLED.show(); delay(1000);
    fill_solid(leds, NUM_LEDS, CRGB(255, 0, 0)); FastLED.show(); delay(1000);
    fill_solid(leds, NUM_LEDS, CRGB(0, 255, 0)); FastLED.show(); delay(1000);
    fill_solid(leds, NUM_LEDS, CRGB(0, 0, 255)); FastLED.show(); delay(1000);
}
